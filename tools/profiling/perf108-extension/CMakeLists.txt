cmake_minimum_required(VERSION 3.5...3.29)
project(dia_rsum LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
  message(FATAL_ERROR "Only Linux x86_64 is qualified for the ordered long-double kernel")
endif()
add_library(dia_rsum SHARED src/dia_rsum.c)
target_include_directories(dia_rsum PRIVATE src)
target_compile_definitions(dia_rsum PRIVATE DUCKDB_EXTENSION_API_VERSION_MAJOR=1 DUCKDB_EXTENSION_API_VERSION_MINOR=2 DUCKDB_EXTENSION_API_VERSION_PATCH=0)
target_compile_options(dia_rsum PRIVATE -O2 -fno-fast-math -Wall -Wextra -Werror)
set_target_properties(dia_rsum PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
